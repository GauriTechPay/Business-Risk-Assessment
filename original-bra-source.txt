const BAND_META = {
  low:    {t:'Low',          c:'b-low',     v:'var(--low)'},
  lowmed: {t:'Low-Medium',   c:'b-lowmed',  v:'var(--lowmed)'},
  med:    {t:'Medium',       c:'b-med',     v:'var(--med)'},
  medhigh:{t:'Medium-High',  c:'b-medhigh', v:'var(--medhigh)'},
  high:   {t:'High',         c:'b-high',    v:'var(--high)'}
};
function bandKey(v){
  if(v<4)  return 'low';
  if(v<8)  return 'lowmed';
  if(v<11) return 'med';
  if(v<20) return 'medhigh';
  return 'high';
}
function band(v){
  const k=bandKey(v);
  return {t:BAND_META[k].t, c:BAND_META[k].c};
}

/* ---------- Gauge (semicircular risk indicator) ---------- */
function polarPt(cx,cy,r,angleDeg){
  const rad=angleDeg*Math.PI/180;
  return {x:cx+r*Math.cos(rad), y:cy+r*Math.sin(rad)};
}
function gaugeArcPath(cx,cy,r,a1,a2){
  const p1=polarPt(cx,cy,r,a1), p2=polarPt(cx,cy,r,a2);
  const large=(a2-a1)>180?1:0;
  return `M ${p1.x.toFixed(2)} ${p1.y.toFixed(2)} A ${r} ${r} 0 ${large} 1 ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`;
}
function angleForValue(v,max){
  const c=Math.max(0,Math.min(max,v));
  return 180 + (c/max)*180;
}
function gaugeSVG(value, max, opts){
  opts = opts||{};
  const cx=90, cy=92, r=72, sw=15;
  const bounds=[0,4,8,11,20,max];
  const keys=['low','lowmed','med','medhigh','high'];
  const bands=keys.map((k,i)=>[bounds[i],bounds[i+1],k]);
  const bandsHTML=bands.map(([s,e,k])=>{
    const a1=angleForValue(s,max), a2=angleForValue(e,max);
    return `<path d="${gaugeArcPath(cx,cy,r,a1,a2)}" fill="none" stroke="${BAND_META[k].v}" stroke-width="${sw}"/>`;
  }).join('');
  const val=Math.max(0,Math.min(max,value));
  const needleAngle=angleForValue(val,max);
  const tip=polarPt(cx,cy,r-sw/2-6,needleAngle);
  const k=bandKey(val);
  const svg=`<svg viewBox="0 0 180 112" class="gauge-svg" role="img" aria-label="${opts.title||''} gauge, value ${value} of ${max}, ${BAND_META[k].t}">
    ${bandsHTML}
    <line x1="${cx}" y1="${cy}" x2="${tip.x.toFixed(2)}" y2="${tip.y.toFixed(2)}" stroke="var(--ink)" stroke-width="3" stroke-linecap="round"/>
    <circle cx="${cx}" cy="${cy}" r="5.5" fill="var(--ink)"/>
    <text x="${cx}" y="${cy-14}" text-anchor="middle" class="gauge-value">${typeof value==='number'&&value%1!==0?value.toFixed(1):value}</text>
  </svg>`;
  if(opts.bare) return svg;
  return `<div class="gauge-box">
    ${svg}
    <div class="gauge-caption">
      ${opts.title?`<div class="gauge-title">${opts.title}</div>`:''}
      <span class="badge ${BAND_META[k].c}">${BAND_META[k].t}</span>
    </div>
  </div>`;
}
function qEV(q, side){
  const imp=q[side+'Imp'], prob=q[side+'Prob'];
  return (imp!==null&&prob!==null)? imp*prob : null;
}
function dimCalc(id){
  const d=S.dims[id];
  let mlEV=0, tfEV=0, mlDone=0, tfDone=0;
  d.q.forEach(q=>{
    const m=qEV(q,'ml'), t=qEV(q,'tf');
    if(m!==null){ mlEV+=m; mlDone++; }
    if(t!==null){ tfEV+=t; tfDone++; }
  });
  const n=d.q.length;
  return {mlEV, tfEV, mlAgg:Math.floor(mlEV/n), tfAgg:Math.floor(tfEV/n), mlDone, tfDone, n};
}
function overallCalc(){
  let mlEV=0, tfEV=0;
  DIMS.forEach(d=>{ const c=dimCalc(d.id); mlEV+=c.mlEV; tfEV+=c.tfEV; });
  const mlTotal=Math.floor(mlEV/25), tfTotal=Math.floor(tfEV/25);
  const overall=mlTotal*0.9 + tfTotal*0.1;
  return {mlEV, tfEV, mlTotal, tfTotal, overall};
}

}

/* ---------- Configure tab: theme logic ---------- */
function applyFullTheme(theme){
  const root=document.documentElement.style;
  root.setProperty('--ink', theme.text);
  root.setProperty('--paper', theme.background);
  root.setProperty('--brand', theme.primary);
  root.setProperty('--secondary', theme.secondary);
  S.configTheme={selected:theme.name, text:theme.text, background:theme.background, primary:theme.primary, secondary:theme.secondary, sidebar:S.configTheme.sidebar};
  renderMain();
}
function selectTheme(name){
  if(!name) return;
  const theme=THEMES.find(t=>t.name===name);
  if(theme) applyFullTheme(theme);
}
function applyManualColor(cssVar,key,inputEl){
  const v=(inputEl.value||'').trim();
  if(!/^#[0-9a-fA-F]{6}$/.test(v)) return; // invalid hex: ignore, leave current color unchanged
  document.documentElement.style.setProperty(cssVar, v);
  S.configTheme[key]=v;
  S.configTheme.selected='';
  const swatchIds={primary:'swatchPrimary',secondary:'swatchSecondary',sidebar:'swatchSidebar'};
  const sw=document.getElementById(swatchIds[key]);
  if(sw) sw.style.background=v;
  const sel=document.getElementById('themeSelect');
  if(sel) sel.value='';
}
function resetTheme(){
  const root=document.documentElement.style;
  ['--ink','--paper','--brand','--secondary','--accent'].forEach(p=>root.removeProperty(p));
  S.configTheme={selected:'', ...DEFAULT_THEME_COLORS};
  renderMain();
}

function levelBandClass(level){
  const map={Low:'b-low','Low-Medium':'b-lowmed',Medium:'b-med','Medium-High':'b-medhigh',High:'b-high'};
  return map[level]||'b-none';
}
function levelBadge(level){
  if(level==='-') return `<span class="badge b-none" style="display:inline-block;padding:3px 9px">-</span>`;

}

/* ---- Page navigation ---- */
function goToFilings(status, ev){ G.filingsStatus = status; return navTo('filings', ev); }
function navTo(page, ev){
  if(ev) ev.preventDefault();
  G.page=page;
  document.querySelectorAll('.app-page').forEach(el=>el.classList.remove('show'));
  const el=document.getElementById('page-'+page);
  if(el) el.classList.add('show');
  document.querySelectorAll('.sb-item[data-page]').forEach(a=>a.classList.toggle('active', a.getAttribute('data-page')===page));
  if(page==='dashboard'){ const m=PAGE_META.dashboard[G.role]; updateHeader(m.title, m.sub); renderDashboard(); }
  else if(page==='filings'){ const m=PAGE_META.filings[G.role]; updateHeader(m.title, m.sub); renderFilings(); }
  else if(page==='assessment'){ updateHeader('New Assessment', assessmentMetaHTML()); }
  else if(page==='reviewform'){ renderReviewForm(); }
  else if(page==='settings'){ updateHeader('Settings', 'Manage your profile, notifications, and appearance'); renderSettingsPage(); }
  else if(page==='admin-mappings'){ updateHeader('Mappings', 'Assign an approver to each preparer - their filings follow automatically'); renderAdminPreparers(); }
  else if(page==='admin-filings'){ updateHeader('Filings', 'Read-only overview of every filing and the approver it routes to'); renderAdminFilings(); }
  else if(page==='admin-sections'){ updateHeader('Sections & Questions', "Add sections and manage questions - admin's own static copy of the form"); renderAdminSections(); }
  else if(page==='admin-guidance'){ updateHeader('Guidance', "Edit the methodology intro and legal framework entries shown on the Guidance step"); renderAdminGuidance(); }
  else if(page==='admin-mapping'){ updateHeader('Risk Mapping', 'Verify and edit the score-to-level band mapping'); renderAdminMapping(); }
  closeProfileMenu();
  toggleNotif(null,false);
  window.scrollTo(0,0);
  return false;
}

/* ---- Assign an approver to a PREPARER. All of that preparer's non-draft filings
   inherit the approver immediately, so nothing needs to be assigned filing-by-filing. ---- */
function openAssignApproverModal(preparerId){
  const p=PREPARERS.find(x=>x.id===preparerId);
  if(!p) return;
  G.assignPreparerId=preparerId;
  const count = FILINGS.filter(f=>f.preparer===p.name && f.reviewStage!=='Draft').length;
  document.getElementById('assignModalBox').innerHTML=`
    <div class="modal-head">
      <div><h3>Assign Approver</h3><div class="msub">${p.name} · ${count} filing${count===1?'':'s'}</div></div>
      <button class="modal-close" onclick="closeAssignModal()">&times;</button>
    </div>
    <div class="modal-body">
      <div class="field">
        <label>Approver</label>
        <select id="assignApproverSelect">
          <option value="">- Choose an approver -</option>
          ${ADMIN_APPROVERS.map(a=>`<option value="${a}" ${p.assignedApprover===a?'selected':''}>${a}</option>`).join('')}
        </select>
      </div>
      <div class="sub-note" style="margin-top:10px">All of ${p.name}'s current and future filings will route to this approver.</div>
    </div>
    <div class="modal-foot">
      <button class="btn-ghost" onclick="closeAssignModal()">Cancel</button>
      <button class="btn-primary" onclick="confirmAssignApprover()">${p.assignedApprover?'Reassign':'Assign'}</button>
    </div>`;
  document.getElementById('assignModalOverlay').classList.add('show');
}
function closeAssignModal(){ document.getElementById('assignModalOverlay').classList.remove('show'); G.assignPreparerId=null; }
function confirmAssignApprover(){
  const sel=document.getElementById('assignApproverSelect');
  const approver=sel?sel.value:'';
  if(!approver){ showToast('Please choose an approver.', 'error'); return; }
  const p=PREPARERS.find(x=>x.id===G.assignPreparerId);
  closeAssignModal();
  if(!p) return;
  const wasUnassigned = !p.assignedApprover;
  p.assignedApprover=approver;
  p.isNew=false;
  let updated=0;
  FILINGS.forEach(f=>{
    if(f.preparer===p.name && f.reviewStage!=='Draft'){
      f.assignedApprover=approver;
      updated++;
      f.audit.unshift({status:'done', title:`${wasUnassigned?'Assigned':'Reassigned'} to ${approver} by Admin`, time:'Just now'});
    }
  });
  const filingNote = updated ? ` - ${updated} filing${updated===1?'':'s'} now routed to them` : '';
  showToast(`${p.name} ${wasUnassigned?'assigned to':'reassigned to'} ${approver}${filingNote}.`, 'success');
  if(G.page==='admin-mappings') renderAdminPreparers();
  else if(G.page==='admin-filings') renderAdminFilings();
  else renderAdminDashboard();
}

/* ---- Filings / Review Queue ---- */
const FILING_STATUSES_PREPARER=['All','Draft','Under Review','Approved','Rejected'];
const FILING_STATUSES_APPROVER=['All','Pending Review','Changes Requested','Approved','Rejected'];
function setFilingsStatus(s){ G.filingsStatus=s; renderFilings(); }
function setFilingsSearch(v){ G.filingsSearch=v; renderFilingsRows(); }
function renderFilings(){
  const statuses = G.role==='approver' ? FILING_STATUSES_APPROVER : FILING_STATUSES_PREPARER;
  document.getElementById('filingsPills').innerHTML = statuses.map(s=>
    `<button class="btn-pill ${G.filingsStatus===s?'active':''}" onclick="setFilingsStatus('${s}')">${s}</button>`).join('');
  const input=document.getElementById('filingsSearchInput');
  if(input) input.value=G.filingsSearch;
  renderFilingsRows();
}
function renderFilingsRows(){
  const q=G.filingsSearch.trim().toLowerCase();
  const isApprover = G.role==='approver';
  let base = isApprover ? FILINGS.filter(f=>f.reviewStage!=='Draft') : FILINGS;
  const rows=base.filter(f=>{
    const stageValue = isApprover ? f.reviewStage : f.status;
    const matchesStatus = G.filingsStatus==='All' || stageValue===G.filingsStatus;
    const matchesSearch = !q || f.entity.toLowerCase().includes(q);
    return matchesStatus && matchesSearch;
  });
  const tbody=document.getElementById('filingsTbody');
  if(!rows.length){
    tbody.innerHTML=`<tr class="empty-row"><td colspan="7">No ${isApprover?'submissions':'filings'} match your search or filters.</td></tr>`;
    return;
  }
  tbody.innerHTML=rows.map(f=>{
    const stageValue = isApprover ? f.reviewStage : f.status;
    return `<tr onclick="openFilingModal('${f.id}')">
      <td>${f.entity}</td>
      <td>${f.period}</td>
      <td><span class="status-badge ${STATUS_CLASS[stageValue]}">${stageValue}</span></td>
      <td class="risk-inline">ML: <b>${f.ml}</b> · TF: <b>${f.tf}</b></td>
      <td>${f.updated}</td>
      <td>${f.preparer}</td>
    </tr>`;
  }).join('');
}

/* ---- Assessment Review Form: full static report (reusing the existing report-generation
   logic) with the audit trail where the live-summary rail normally sits. Decision buttons
   (Approve / Request Changes / Reject) live here, and only show while a filing is still
   undecided - once Approved or Rejected, the filing is final and read-only. ---- */
function openReviewForm(id){
  const f=FILINGS.find(x=>x.id===id);
  if(!f) return;
  closeFilingModal();
  G.reviewFilingId=id;
  navTo('reviewform');
}
function renderReviewForm(){
  const f=FILINGS.find(x=>x.id===G.reviewFilingId);
  if(!f){ navTo('filings'); return; }
  updateHeader(f.entity, `${f.type} Assessment · ${f.period} · Prepared by ${f.preparer}`);
  document.getElementById('reviewFormMain').innerHTML = PAGES[7]();
  document.getElementById('reviewFormTimeline').innerHTML = buildTimelineHTML(f);
  document.getElementById('reviewScroll').scrollTop = 0;
  const decided = f.reviewStage==='Approved' || f.reviewStage==='Rejected';
  const actionsEl = document.getElementById('reviewFormActions');
  const backBtn = `<button class="btn-ghost" style="margin-right:auto" onclick="navTo('filings')">← Back to Review Queue</button>`;
  actionsEl.style.display='flex';
  if(decided){
    actionsEl.innerHTML = backBtn;
  } else {
    actionsEl.innerHTML = backBtn + `
      <button class="btn-warn" onclick="openActionModal('changes','${f.id}')">Request Changes</button>
      <button class="btn-danger" onclick="openActionModal('reject','${f.id}')">Reject</button>
      <button class="btn-success" onclick="approverApprove('${f.id}')">Approve</button>`;
  }
}

/* ---- Approver decisions: Approve is immediate; Request Changes / Reject collect a comment first.
   Static demo data, but the in-memory filing record is updated so the queue/dashboard reflect
   the decision and a decided filing stops offering further actions. ---- */
function approverApprove(id){
  const f=FILINGS.find(x=>x.id===id);
  if(!f) return;
  closeFilingModal();
  f.reviewStage='Approved';
  f.audit.unshift({status:'done', title:'Approved by Kiran Alvarez', time:'Just now'});
  showToast(`${f.entity} approved.`, 'success');
  navTo('filings');
}
function openActionModal(kind, filingId){
  const f=FILINGS.find(x=>x.id===filingId);
  if(!f) return;
  closeFilingModal();
  G.actionKind=kind;
  G.actionFilingId=filingId;
  const isReject = kind==='reject';
  const title = isReject ? 'Reject Assessment' : 'Request Changes';
  const label = isReject ? 'Reason for rejection' : 'What needs to change?';
  const placeholder = isReject ? 'Explain why this assessment cannot be approved…' : 'Describe what the preparer needs to revise…';
  const submitLabel = isReject ? 'Reject Assessment' : 'Send Request';
  const submitClass = isReject ? 'btn-danger' : 'btn-warn';
  document.getElementById('actionModalBox').innerHTML=`
    <div class="modal-head">
      <div><h3>${title}</h3></div>
      <button class="modal-close" onclick="closeActionModal()">&times;</button>
    </div>
    <div class="modal-body">
      <div class="action-context">${f.entity} - ${f.type} Assessment · ${f.period}</div>
      <div class="action-field">
        <label>${label}</label>
        <textarea id="actionComment" placeholder="${placeholder}"></textarea>
      </div>
    </div>
    <div class="modal-foot">
      <button class="btn-ghost" onclick="closeActionModal()">Cancel</button>
      <button class="${submitClass}" onclick="submitActionModal()">${submitLabel}</button>
    </div>`;
  document.getElementById('actionModalOverlay').classList.add('show');
}
function closeActionModal(){
  document.getElementById('actionModalOverlay').classList.remove('show');
  G.actionKind=null;
  G.actionFilingId=null;
}
function submitActionModal(){
  const ta=document.getElementById('actionComment');
  const text=ta?ta.value.trim():'';
  if(!text){
    showToast('Please add a comment before submitting.', 'error');
    return;
  }
  const f=FILINGS.find(x=>x.id===G.actionFilingId);
  const kind=G.actionKind;
  closeActionModal();
  if(!f) return;
  if(kind==='reject'){
    f.reviewStage='Rejected';
    f.audit.unshift({status:'rejected', title:'Rejected by Kiran Alvarez', note:text, time:'Just now'});
    showToast(`${f.entity} rejected.`, 'success');
  } else {
    f.reviewStage='Changes Requested';
    f.audit.unshift({status:'stuck', title:'Changes requested by Kiran Alvarez', note:text, time:'Just now'});
    showToast(`Changes requested - feedback sent to the preparer.`, 'success');
  }
  navTo('filings');
}

/* ---- Toasts: stack top-right, auto-dismiss after 4s, independent per-toast timers ---- */
function showToast(message, type){
  const stack=document.getElementById('toastStack');
  const el=document.createElement('div');
  el.className='toast '+(type==='error'?'error':'success');
  const icon = type==='error'
    ? '<svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4"><circle cx="12" cy="12" r="10"/><path d="M12 8v5"/><path d="M12 16h.01"/></svg>'
    : '<svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4"><path d="M20 6L9 17l-5-5"/></svg>';
  el.innerHTML = icon + `<span>${message}</span>`;
  stack.appendChild(el);
  requestAnimationFrame(()=>el.classList.add('show'));
  setTimeout(()=>{
    el.classList.remove('show');
    setTimeout(()=>{ el.remove(); }, 200);
  }, 4000);
}
